    function userAdjustFunds(userId, mode) {
      const u = state.users.find(x => x.id === userId);
      const amt = parseFloat(document.getElementById('fund-input-' + userId).value);
      if (!u || !amt || amt <= 0) {
        alert("Enter a valid amount.");
        return;
      }
      u.balance = mode === 'add' ? u.balance + amt : Math.max(0, u.balance - amt);
      saveState();
      alert(`${mode === 'add' ? 'Added' : 'Deducted'} $${amt.toFixed(2)} for ${u.name}`);
    }

    function saveUserLuckyOrder(userId) {
      const u = state.users.find(x => x.id === userId);
      if (!u) return;
      const tNum = parseInt(document.getElementById('u-lucky-task-' + userId).value);
      const prodSplit = document.getElementById('u-lucky-prod-' + userId).value.split('|');
      const amt = parseFloat(document.getElementById('u-lucky-amt-' + userId).value);
      const comm = parseFloat(document.getElementById('u-lucky-comm-' + userId).value);
      if (!tNum || tNum < 1 || tNum > 25 || !amt || !comm) {
        alert("Enter valid task number, amount, and commission.");
        return;
      }
      u.activeLuckyOrder = {
        taskNum: tNum,
        product: prodSplit[0],
        img: prodSplit[1],
        amount: amt,
        comm: comm,
        triggered: u.completedTasks + 1 === tNum,
        completed: false
      };
      saveState();
      alert(`Lucky Order set for ${u.name} on Task #${tNum}!`);
    }

    function handleCustLogin() {
      const email = (document.getElementById('login-email').value.trim() || "sara.khan@email.com").toLowerCase();
      const pass = document.getElementById('login-pass').value;
      let existing = state.users.find(x => x.email.toLowerCase() === email);

      if (existing) {
        if (existing.loginPassword && pass !== existing.loginPassword) {
          alert("Incorrect Login Password!");
          return;
        }
        if (existing.accountStatus === "Deactive") {
          alert("Your account is Deactivated. Please contact customer service.");
          return;
        }
        state.activeUserId = existing.id;
      } else {
        const created = makeUserRecord({
          name: email.split('@')[0].replace(/[._]/g, ' ').toUpperCase(),
          phone: "+971 50 000 0000",
          email: email,
          loginPassword: pass
        });
        state.users.unshift(created);
        state.activeUserId = created.id;
      }

      state.customerLoggedIn = true;
      saveState();
      applyRouting();
    }

    function grabOrder() {
      const u = getActiveUser();
      const lucky = u.activeLuckyOrder;

      if (u.accountStatus !== "Active") {
        alert(`Action blocked: Account is ${u.accountStatus}.`);
        return;
      }
      if (u.tasksLocked) {
        alert("Please contact with customer service");
        return;
      }
      if (lucky && lucky.triggered && !lucky.completed) {
        openCustModal('deposit1');
        return;
      }
      if (u.completedTasks >= u.totalTasks) {
        alert("All 25 unique tasks completed!");
        return;
      }

      const nextTaskNum = u.completedTasks + 1;
      const btn = document.getElementById('grab-btn');
      if (btn) {
        btn.disabled = true;
        btn.innerHTML = '<i class="fa-solid fa-spinner fa-spin"></i> Processing...';
      }

      setTimeout(() => {
        if (lucky && !lucky.completed && lucky.taskNum === nextTaskNum) {
          lucky.triggered = true;
          if (u.balance < lucky.amount) {
            saveState();
            return;
          } else {
            completeLuckyOrder();
            return;
          }
        }

        const item = unique25Products[u.completedTasks % unique25Products.length];
        u.completedTasks++;
        u.balance += item.comm;
        u.todayEarn += item.comm;

        state.ordersHistory.unshift({
          userId: u.id,
          taskNum: nextTaskNum,
          name: item.name,
          store: item.store,
          comm: item.comm,
          img: item.img
        });

        saveState();
      }, 450);
    }

    function completeLuckyOrder() {
      const u = getActiveUser();
      const lucky = u.activeLuckyOrder;
      if (!lucky || u.balance < lucky.amount) {
        openCustModal('deposit1');
        return;
      }

      u.completedTasks++;
      u.balance += lucky.comm;
      u.todayEarn += lucky.comm;
      lucky.completed = true;

      state.ordersHistory.unshift({
        userId: u.id,
        taskNum: lucky.taskNum,
        name: lucky.product + " (Lucky)",
        store: "VIP Lucky",
        comm: lucky.comm,
        img: lucky.img
      });

      saveState();
      alert(`Lucky Order Completed! +$${lucky.comm.toFixed(2)} commission credited.`);
    }

    function confirmPaid() {
      const u = getActiveUser();
      const now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      state.txHistory.unshift({
        id: Date.now(),
        userId: u.id,
        user: u.name,
        type: "Deposit",
        amount: pendingDepositAmt,
        status: "Pending",
        time: "Today, " + now
      });
      saveState();
      alert("Deposit submitted live! Waiting for Admin approval.");
      closeModal();
    }

    function submitWithdraw() {
      const u = getActiveUser();
      if (u.withdrawBlocked) {
        alert("Withdrawals are BLOCKED on your account.");
        return;
      }
      const amt = parseFloat(document.getElementById('wd-amt-input').value);
      const wdPass = document.getElementById('wd-pass-input').value;
      if (u.withdrawPassword && wdPass !== u.withdrawPassword) {
        alert("Incorrect Withdraw Password!");
        return;
      }
      if (!amt || amt <= 0 || amt > u.balance) {
        alert("Invalid or insufficient balance.");
        return;
      }
      u.balance -= amt;
      const now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
      state.txHistory.unshift({
        id: Date.now(),
        userId: u.id,
        user: u.name,
        type: "Withdraw",
        amount: amt,
        status: "Pending",
        time: "Today, " + now
      });
      saveState();
      alert("Withdrawal submitted live! Waiting for Admin approval.");
      closeModal();
    }

    function saveBind() {
      const u = getActiveUser();
      const t = document.getElementById('bind-type').value;
      const v = document.getElementById('bind-val').value.trim();
      if (!v) return;
      u.boundWallet = `${t}: ${v.slice(0, 12)}...`;
      saveState();
      closeModal();
    }

    function handleAdminLogin() {

    function decideAdminTx(id, decision) {
      const tx = state.txHistory.find(t => t.id === id);
      if (!tx || tx.status !== 'Pending') return;
      const targetUser = state.users.find(x => x.id === tx.userId) || getActiveUser();

      tx.status = decision;
      if (tx.type === 'Deposit' && decision === 'Approved') {
        targetUser.balance += tx.amount;
      }
      if (tx.type === 'Withdraw' && decision === 'Rejected') {
        targetUser.balance += tx.amount;
      }

      saveState();
      alert(`${tx.type} ${decision} live!`);
    }

    }
